def meshpart(name,verts,faces,mat,bone,bevel=.006):
    me=bpy.data.meshes.new(name);me.from_pydata(verts,[],faces);me.update()

    return finish(o,name,mat,bone,bevel)

def plate(name,outline,y,depth,mat,bone,bevel=.006):
    # outline: x,z in perimeter order
    n=len(outline);verts=[(x,y,z) for x,z in outline]+[(x,y+depth,z) for x,z in outline]
    faces=[tuple(reversed(range(n))),tuple(range(n,2*n))]
    faces += [(i,(i+1)%n,(i+1)%n+n,i+n) for i in range(n)]
    return meshpart(name,verts,faces,mat,bone,bevel)

def loft(name,xc,sections,mat,bone,bevel=.009):
    # Chamfered rectangular sections: (z,width,depth,y-center)
    verts=[]
    for z,w,d,y in sections:
        c=min(w,d)*.18
        for x,yy in [(-w/2+c,-d/2),(w/2-c,-d/2),(w/2,-d/2+c),(w/2,d/2-c),
                     (w/2-c,d/2),(-w/2+c,d/2),(-w/2,d/2-c),(-w/2,-d/2+c)]:
            verts.append((xc+x,y+yy,z))
    faces=[tuple(reversed(range(8))),tuple(range(len(verts)-8,len(verts)))]
    for j in range(len(sections)-1):
        for i in range(8):faces.append((j*8+i,j*8+(i+1)%8,(j+1)*8+(i+1)%8,(j+1)*8+i))
    return meshpart(name,verts,faces,mat,bone,bevel)
