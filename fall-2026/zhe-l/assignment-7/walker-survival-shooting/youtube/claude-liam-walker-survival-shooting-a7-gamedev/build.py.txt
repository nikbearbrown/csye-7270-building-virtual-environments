# ---------- drawing helpers ----------
def canvas():
    im = Image.new("RGB", (W, H), BG); d = ImageDraw.Draw(im)
    d.rectangle([0, 0, W, 14], fill=INK)
    return im, d

def tag(d, text, x=W - 40, y=40, fill=RED):
    f = F(BOLD, 30); w = d.textlength(text, font=f)
    d.rectangle([x - w - 28, y, x, y + 50], fill=fill); d.text((x - w - 14, y + 5), text, font=f, fill=BG)

def title(d, t, sub=None):
    d.text((80, 50), t, font=F(BOLD, 64), fill=INK)
    if sub: d.text((84, 135), sub, font=F(REG, 32), fill=GREY)

def foot(d, t):
    d.text((80, H - 60), t, font=F(REG, 24), fill=GREY)

def wrap(d, text, font, maxw):
    out, line = [], ""
    for w in text.split():
        t = (line + " " + w).strip()
        if d.textlength(t, font=font) > maxw and line: out.append(line); line = w
        else: line = t
    if line: out.append(line)
    return out

def bullets(d, items, x=80, y=220, size=40, maxw=1760, gap=18):
    f = F(REG, size)
    for it in items:
        bold = it.startswith("!")
        it = it.lstrip("!")
        ff = F(BOLD, size) if bold else f
        for k, ln in enumerate(wrap(d, it, ff, maxw - 40)):
            if k == 0: d.rectangle([x, y + size * 0.45, x + 14, y + size * 0.45 + 14], fill=RED)
            d.text((x + 40, y), ln, font=ff, fill=INK); y += int(size * 1.3)
        y += gap
    return y

def code(d, rows, x, y, w, size=26, hl=()):
    f = F(MONO, size); lh = int(size * 1.45)
    h = lh * len(rows) + 30
    d.rectangle([x, y, x + w, y + h], fill=CODEBG)
    maxc = int((w - 110) / d.textlength("M", font=f))
    for k, (n, s) in enumerate(rows):
        yy = y + 15 + k * lh
        if n in hl: d.rectangle([x, yy - 2, x + w, yy + lh - 4], fill=(70, 40, 40))
        d.text((x + 12, yy), f"{n:>4}", font=f, fill=(120, 120, 140))
        s = s.replace("\t", "    ")
        d.text((x + 90, yy), s[:maxc] + ("…" if len(s) > maxc else ""), font=f, fill=CODEFG)
    return y + h

CROP = (640, 470, 1080, 1010)
def paste_fit(im, path, box, nearest=False, crop=None):
    src = Image.open(path).convert("RGB")
    if crop: src = src.crop(crop)
    x0, y0, x1, y1 = box; bw, bh = x1 - x0, y1 - y0
    s = min(bw / src.width, bh / src.height)
    src = src.resize((int(src.width * s), int(src.height * s)), Image.NEAREST if nearest else Image.LANCZOS)
    im.paste(src, (x0 + (bw - src.width) // 2, y0 + (bh - src.height) // 2))
    return src.size

def tts():
    from kokoro_onnx import Kokoro
    m = Path(r"<brutalist.art>\runtime\models\kokoro")
    k = Kokoro(str(m / "kokoro-v1.0.onnx"), str(m / "voices-v1.0.bin"))
    for b in BEATS:
        out = AUD / f"{b['id']}.wav"
        if out.exists() and json.loads((AUD / f"{b['id']}.json").read_text()).get("text") == b["text"]: continue
        s, sr = k.create(b["text"], voice="am_onyx", speed=1.0, lang="en-us")
        pad = np.zeros(int(sr * 0.35), dtype=np.float32)
        s = np.concatenate([pad, s.astype(np.float32), pad * 2])
        with wave.open(str(out), "wb") as w:
            w.setnchannels(1); w.setsampwidth(2); w.setframerate(sr)
            w.writeframes((np.clip(s, -1, 1) * 32767).astype(np.int16).tobytes())
        (AUD / f"{b['id']}.json").write_text(json.dumps({"text": b["text"], "sec": len(s) / sr}))
        print("tts", b["id"], round(len(s) / sr, 2), flush=True)

def dur(p):
    with wave.open(str(p)) as w: return w.getnframes() / w.getframerate()

def run(cmd): subprocess.run(cmd, check=True, capture_output=True)

def cards():
    for b in BEATS:
        if b["kind"] != "card": continue
        im, d = canvas(); R[b["draw"]](im, d)
        im.save(CARDS / f"{b['id']}.png")

def label_png(text, path, color=RED):
    im = Image.new("RGBA", (W, H), (0, 0, 0, 0)); d = ImageDraw.Draw(im)
    f = F(BOLD, 34); w = d.textlength(text, font=f)
    d.rectangle([40, 30, 40 + w + 30, 86], fill=color + (235,)); d.text((55, 36), text, font=f, fill=(255, 255, 255))
    im.save(path)

def clips():
    out = []
    for b in BEATS:
        a = AUD / f"{b['id']}.wav"; t = dur(a); o = CLIPS / f"{b['id']}.mp4"
        enc = ["-c:v", "libx264", "-preset", "veryfast", "-crf", "20", "-pix_fmt", "yuv420p", "-r", "30", "-c:a", "aac", "-b:a", "192k", "-ar", "48000", "-ac", "2"]
        if b["kind"] == "card":
            run([FFMPEG, "-y", "-loop", "1", "-framerate", "30", "-i", str(CARDS / f"{b['id']}.png"), "-i", str(a), "-t", f"{t:.3f}", *enc, str(o)])
        else:
            cap = WORK / "captures/play-01.mp4"
            tl = WORK / "captures/labels.json"
            vd = float(subprocess.run(["ffprobe", "-v", "error", "-show_entries", "format=duration", "-of", "csv=p=0", str(cap)], capture_output=True, text=True).stdout.strip())
            T = max(t, vd)
            label_png("SCRIPTED-INPUT CAPTURE · Tab via Input.parse_input_event · camera turned 180° at the start", CLIPS / "lab_main.png")
            inputs = ["-i", str(cap), "-i", str(CLIPS / "lab_main.png")]
            fc = f"[0:v]scale={W}:{H},fps=30,tpad=stop_mode=clone:stop_duration={max(0, T - vd) + 0.1:.2f}[v0];[v0][1:v]overlay=0:0[v1]"
            last = "v1"
            if tl.exists():
                segs = json.loads(tl.read_text())
                for i, (s0, s1, txt) in enumerate(segs):
                    p = CLIPS / f"lab_{i}.png"
                    im = Image.new("RGBA", (W, H), (0, 0, 0, 0)); dd = ImageDraw.Draw(im); f = F(BOLD, 44)
                    w = dd.textlength(txt, font=f); dd.rectangle([40, H - 130, 40 + w + 40, H - 50], fill=(17, 17, 17, 230)); dd.text((60, H - 122), txt, font=f, fill=(255, 255, 255))
                    im.save(p); inputs += ["-i", str(p)]
                    fc += f";[{last}][{i + 2}:v]overlay=0:0:enable='between(t,{s0},{s1})'[o{i}]"; last = f"o{i}"
            if T > vd + 0.05:
                p = CLIPS / "lab_held.png"; label_png("HELD FRAME", p, INK); inputs += ["-i", str(p)]
                n = len(inputs) // 2 - 1
                fc += f";[{last}][{n}:v]overlay=0:100:enable='gte(t,{vd:.2f})'[oh]"; last = "oh"
            na = len(inputs) // 2
            fc += f";[{na}:a]apad=whole_dur={T:.3f}[a]"
            run([FFMPEG, "-y", *inputs, "-i", str(a), "-filter_complex", fc, "-map", f"[{last}]", "-map", "[a]", "-t", f"{T:.3f}", *enc, str(o)])
        out.append(o); print("clip", b["id"], flush=True)
    lst = B / "concat.txt"; lst.write_text("".join(f"file '{p.as_posix()}'\n" for p in out))
    final = Path(r"<film-out>\claude-liam-walker-survival-shooting-a7-gamedev-1080p.mp4")
    run([FFMPEG, "-y", "-f", "concat", "-safe", "0", "-i", str(lst), "-c", "copy", "-movflags", "+faststart", str(final)])
    print("FINAL", final)
