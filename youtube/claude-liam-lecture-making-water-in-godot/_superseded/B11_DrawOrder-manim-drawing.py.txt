def a1_clip(subject, clipper):
    """Sutherland-Hodgman: clip a polygon (list of (x, y)) against a convex counter-clockwise polygon."""
    def cross(o, a, b):
        return (a[0] - o[0]) * (b[1] - o[1]) - (a[1] - o[1]) * (b[0] - o[0])

    def inter(p, q, a, b):
        d1 = (q[0] - p[0], q[1] - p[1]); d2 = (b[0] - a[0], b[1] - a[1])
        den = d1[0] * d2[1] - d1[1] * d2[0]
        t = ((a[0] - p[0]) * d2[1] - (a[1] - p[1]) * d2[0]) / den
        return (p[0] + t * d1[0], p[1] + t * d1[1])
    out = list(subject)
    for i in range(len(clipper)):
        a, b = clipper[i], clipper[(i + 1) % len(clipper)]
        inp, out = out, []
        if not inp:
            break
        s = inp[-1]
        for e in inp:
            ine, ins = cross(a, b, e) >= 0, cross(a, b, s) >= 0
            if ine:
                if not ins:
                    out.append(inter(s, e, a, b))
                out.append(e)
            elif ins:
                out.append(inter(s, e, a, b))
            s = e
    return out


def a1_poly(pts, color, fill, fop, sw=4):
    return Polygon(*[[p[0], p[1], 0] for p in pts], stroke_color=color, stroke_width=sw, fill_color=fill, fill_opacity=fop)


class B11_DrawOrder(Scene):
    def construct(self):
        self.camera.background_color = STAGE
        # top half: solids first, water after
        sand = a1_poly([(-3.3, 1.05), (2.5, 1.05), (3.2, 2.0), (-2.6, 2.0)], INK, a1_SAND, 1.0, 3)
        sand_edge = a1_poly([(-3.3, 1.05), (2.5, 1.05), (2.5, 0.9), (-3.3, 0.9)], INK, a1_SANDDK, 1.0, 3)
        rock = a1_poly([(-1.9, 1.6), (-1.5, 2.05), (-0.9, 2.15), (-0.5, 1.8), (-0.8, 1.45), (-1.5, 1.4)], INK, a1_ROCK, 1.0, 3)
        buoy = VGroup(Line([1.4, 2.35, 0], [1.4, 2.75, 0], color=INK, stroke_width=4), Circle(radius=0.38, stroke_color=INK, stroke_width=3, fill_color=TEAL, fill_opacity=1).move_to([1.4, 2.0, 0]),
                      Dot([1.4, 2.8, 0], radius=0.07, color=CARD).set_stroke(INK, width=2))
        water = a1_poly([(-3.9, 0.8), (2.9, 0.8), (3.9, 2.75), (-2.9, 2.75)], a1_BLUE, a1_BLUE, 0.3, 5)
        l_solid = T("solid first", 36).move_to([-3.4, 0.4, 0]); l_trans = T("transparent after", 36).move_to([0.3, 0.4, 0])
        l_water = T("water", 36).move_to([0.0, 3.05, 0])
        div = Line([-6.0, -0.1, 0], [6.0, -0.1, 0], color=GHOST, stroke_width=3)
        # bottom half: two overlapping translucent planes, each with a dot at its own centre
        A = [(-4.5, -2.2), (-0.1, -2.2), (0.7, -0.5), (-3.7, -0.5)]
        B = [(-1.1, -2.2), (4.5, -2.2), (3.7, -0.5), (-1.9, -0.5)]
        cA = (sum(p[0] for p in A) / 4.0, sum(p[1] for p in A) / 4.0); cB = (sum(p[0] for p in B) / 4.0, sum(p[1] for p in B) / 4.0)
        pB = a1_poly(B, TEAL, TEAL, 0.3, 5); pA = a1_poly(A, a1_BLUE, a1_BLUE, 0.3, 5)
        lens_pts = a1_clip(A, B)
        lens = a1_poly(lens_pts, a1_BLUE, a1_BLUE, 0.6, 3)
        dB = a1_tdot(cB, 0.15); dA = a1_tdot(cA, 0.15)
        l_sort = T("sorted by position", 36).move_to([0.0, -2.85, 0])
        # the wrong part: where the other plane is really nearer, but the nearer-centre plane is painted on top
        L0, L1 = (-0.7, -2.3), (-0.4, -0.4)
        right = a1_clip(lens_pts, [L0, (5.0, -2.3), (5.0, -0.4), L1])
        patch = a1_poly(right, a1_BLUE, a1_BLUE, 0.0, 0)
        depth_line = DashedLine([L0[0], L0[1], 0], [L1[0], L1[1], 0], color=DIM, stroke_width=5, dash_length=0.14)
        wring = ring(patch, pad=0.12, sw=7)
        tag = T("drawing", 32, DIM); a1_left(tag, -6.0, -3.15)

        self.play(Create(div), FadeIn(tag), run_time=0.4)
        # solids first
        until(self, "Godot draws transparent", lead=0.3)
        self.play(LaggedStart(FadeIn(VGroup(sand_edge, sand)), FadeIn(rock, scale=0.7), FadeIn(buoy, scale=0.7), lag_ratio=0.35), FadeIn(l_solid), run_time=0.9)
        # then the translucent water slab over them
        until(self, "after solid ones", lead=0.6)
        self.play(Create(water), run_time=0.7)
        self.play(FadeIn(l_water), FadeIn(l_trans), run_time=0.4)
        # two planes, each with its own centre dot, drawn in order
        until(self, "sorted back to front", lead=0.55)
        self.play(FadeIn(pB), FadeIn(dB, scale=1.6), run_time=0.4)
        self.play(FadeIn(pA), FadeIn(lens), FadeIn(dA, scale=1.6), FadeIn(l_sort), run_time=0.5)
        # --- after the 45 to 55 percent window ---
        until(self, "Two overlapping", lead=0.2)
        self.play(Indicate(lens, color=TEAL, scale_factor=1.0), run_time=0.7)
        until(self, "wrong order", lead=1.3)
        self.play(Create(depth_line), run_time=0.5)
        self.play(FadeIn(wring, scale=1.3), run_time=0.5)
        self.play(*a1_pulse(wring, 1.1), run_time=0.5)
        done(self)
